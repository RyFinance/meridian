+def nth_last_session_of_month(year: int, month: int, n: int) -> dt.date:
+    """The n-th-to-last NYSE session of the month (n=1 -> last session, n=4 -> 4th-to-last).
+
+    Needed for rules like tsy_month_end_v1 ("hold from the close of the
+    4th-to-last session to the close of the last session"), which
+    first_session_of_month cannot express.
+    """
+    if n < 1:
+        raise ValueError("n must be >= 1")
+    start = dt.date(year, month, 1)
+    end = dt.date(year + (month == 12), month % 12 + 1, 1) - dt.timedelta(days=1)
+    sessions = sessions_between(start, end)
+    if n > len(sessions):
+        raise ValueError(f"{year}-{month:02d} has only {len(sessions)} sessions; n={n} is out of range")
+    return sessions[-n]
